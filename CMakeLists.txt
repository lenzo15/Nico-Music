cmake_minimum_required(VERSION 3.18)
# Keep this version aligned with the vX.Y.Z tag when publishing a release.
project(NicoMusic VERSION 1.0.3 LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "NicoMusic targets Windows.")
endif()
add_library(audio STATIC src/audio/player.cpp src/plugins/sources.cpp)
target_compile_features(audio PUBLIC cxx_std_17)
target_compile_definitions(audio PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE)
target_include_directories(audio PUBLIC third_party)
target_link_libraries(audio PUBLIC shell32)
if(MSVC)
  target_compile_options(audio PUBLIC /utf-8 /W4 /permissive-)
elseif(MINGW)
  target_compile_options(audio PUBLIC -Wall -Wextra)
  target_link_options(audio PUBLIC -static)
endif()
add_executable(NicoMusicCLI src/cli/main.cpp)
target_link_libraries(NicoMusicCLI PRIVATE audio)
option(NICOMUSIC_BUNDLE_PLUGINS "Stage plugins listed in the bundled catalog" ON)
if(NICOMUSIC_BUNDLE_PLUGINS)
  include(plugins/bundle.cmake)
  add_dependencies(NicoMusicCLI copy_source_plugins)
endif()
option(NICOMUSIC_SOURCE_TESTS "Build the source plugin protocol test harness" OFF)
if(NICOMUSIC_SOURCE_TESTS)
  add_executable(NicoMusicSourceTest tests/source_page_native.cpp)
  target_link_libraries(NicoMusicSourceTest PRIVATE audio shlwapi)
  if(MINGW)
    target_link_options(NicoMusicSourceTest PRIVATE -municode)
  endif()
endif()

set(desktop_default ${MSVC})
if(DEFINED BILIMUSIC_DESKTOP) # Honor existing build configurations.
  set(desktop_default ${BILIMUSIC_DESKTOP})
endif()
option(NICOMUSIC_DESKTOP "Build the graphical music player (MSVC x64)" ${desktop_default})
if(NICOMUSIC_DESKTOP)
  if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Use MSVC x64 or configure -DNICOMUSIC_DESKTOP=OFF for the CLI only.")
  endif()
  if(POLICY CMP0135)
    cmake_policy(SET CMP0135 NEW)
  endif()
  include(FetchContent)
  FetchContent_Declare(webview2 URL https://api.nuget.org/v3-flatcontainer/microsoft.web.webview2/1.0.2903.40/microsoft.web.webview2.1.0.2903.40.nupkg
    URL_HASH SHA256=ef128016dd1e51c59178c827ed5b8aa3322c57afa8675d930f8109505542ad74 DOWNLOAD_NAME webview2.zip)
  FetchContent_MakeAvailable(webview2)
  set(sdk "${webview2_SOURCE_DIR}/build/native")
  add_executable(NicoMusic WIN32 src/desktop/desktop.cpp src/library/nicomix.cpp src/updates/updater.cpp src/desktop/hotkeys.cpp src/desktop/app.rc)
  if(NICOMUSIC_BUNDLE_PLUGINS)
    add_dependencies(NicoMusic copy_source_plugins)
  endif()
  target_include_directories(NicoMusic SYSTEM PRIVATE "${sdk}/include")
  target_compile_definitions(NicoMusic PRIVATE NICOMUSIC_VERSION="${PROJECT_VERSION}")
  target_compile_options(NicoMusic PRIVATE /EHsc)
  target_link_libraries(NicoMusic PRIVATE audio mmdevapi ole32 windowsapp shlwapi version winhttp bcrypt "${sdk}/x64/WebView2LoaderStatic.lib")
  option(NICOMUSIC_UPDATE_TESTS "Build the native updater test harness" OFF)
  if(NICOMUSIC_UPDATE_TESTS)
    add_executable(NicoMusicUpdateTest tests/updater_native.cpp src/updates/updater.cpp src/desktop/app.rc)
    target_link_libraries(NicoMusicUpdateTest PRIVATE audio winhttp bcrypt ole32)
    target_compile_options(NicoMusicUpdateTest PRIVATE /EHsc)
  endif()
  option(NICOMUSIC_HISTORY_TESTS "Build native library undo/redo regression tests" OFF)
  if(NICOMUSIC_HISTORY_TESTS)
    add_executable(NicoMusicHistoryTest tests/history_native.cpp)
    target_link_libraries(NicoMusicHistoryTest PRIVATE audio)
    target_compile_options(NicoMusicHistoryTest PRIVATE /EHsc)
  endif()
  option(NICOMUSIC_HOTKEY_TESTS "Build native hotkey settings regression tests" OFF)
  if(NICOMUSIC_HOTKEY_TESTS)
    add_executable(NicoMusicHotkeyTest tests/hotkeys_native.cpp src/desktop/hotkeys.cpp)
    target_link_libraries(NicoMusicHotkeyTest PRIVATE audio)
    target_compile_options(NicoMusicHotkeyTest PRIVATE /EHsc)
  endif()
  add_custom_target(copy_ui ALL COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/ui" "$<TARGET_FILE_DIR:NicoMusic>/ui")
  add_custom_command(TARGET NicoMusic POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/ui" "$<TARGET_FILE_DIR:NicoMusic>/ui")
endif()
